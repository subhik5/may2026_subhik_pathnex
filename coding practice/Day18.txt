#🔹 Kubernetes — Node Affinity

#   In values: - ssd containers: - name: web image: nginx

apiVersion: v1
kind: Pod
metadata: 
  name: pathnex-affinity
spec:
  affinity:
    nodeAffinity:
      requiredDuringSchedulingIgnoredDuringExecution
        nodeSelectorTerms:
          - matchExpressions:
            - key: disktype
              operator: In
              values:
              - ssd
  containers:
  - name: web
    image: nginx

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('deploy') {
            steps {
                script {
                    try {
                        sh 'echo "Deploying $INSTITUTE_NAME application..'
                        sh 'exit 1' // intentional failure simulation
                    } catch (Exception e) {
                        echo "Deployment failed. Rolling back $INSTITUTE_NAME application.."
                    }
                }
            }
        }
    }
}

#🔹 GitLab CI — Rollback Simulation
#You will learn how to simulate rollback using after_script.

#stages:

#variables: INSTITUTE_NAME: "Pathnex"
#deploy: stage: deploy image: alpine:latest script:
# - echo "Deploying $INSTITUTE_NAME application..." - exit 1 after_script: - echo 
# "Deployment failed. Rolling back $INSTITUTE_NAME application..."


stages:
  - deploy
variables:
  INSTITUTE_NAME: "Pathnex"
deploy:
  stage: deploy
  image: alpine:latest 
  script:
    - echo "Deploying $INSTITUTE_NAME application..."
    - exit 1 
  after_script:
    - echo "Deployment failed. Rolling back $INSTITUTE_NAME application..."
