#Matrix Builds
#🔹 Jenkins Pipeline — Matrix Build
#You will learn how to run the same build on multiple environments.

#pipeline { agent any environment { INSTITUTE_NAME = "Pathnex" } matrix 
#{ axes { axis { name 'JAVA_VERSION' values '8', '11', '17' } } stages 
#{ stage('Build') { steps { sh "echo Building with Java $JAVA_VERSION" } 
#} } } }

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('Matrix Build Execution') {
            matrix {
                axes {
                    axis {
                        name 'JAVA_VERSION'
                        values '8', '11', '17'
                    }
                }
                stages {
                    stage('Build') {
                        steps {
                            echo "Building with Java ${JAVA_VERSION} under ${INSTITUTE_NAME} runtime"
                            sh "echo Current environment Java version: ${JAVA_VERSION}"
                        }
                    }
                }
            }
        }
    }
}

stages:
  - build

matrix-build:
  stage: build
  image: maven:3.8.1-jdk-17
  parallel:
    matrix:
      - JAVA_VERSION: ['8', '11', '17']
  script:
    - echo "Building and compiling validation tests with Java $JAVA_VERSION"
