provider "aws" {
    region = "us-east-1"
}
resource "aws_key_pair" "pathnexKey" {
    key_name = "pathnexKey"
    public_key = "ssh-rsa AAAA...."
}
resource "aws_instance" "pathenexEC2" {
    ami = "ami-0abcd1234abcd1234" instance_type = "c6i.8xlarge"
    key_name = aws_key_pair.pathnexKey.key_name
    tags = {
        Name = "pathnex-EC2"
    }
}

#Parallel Builds
#🔹 Jenkins Pipeline — Parallel Build & Test
#You will learn how to run build and test in parallel.
#pipeline { agent any stages { stage('Parallel Tasks')
# { parallel { stage('Build') { steps { sh 'mvn clean compile' } 
# } stage('Test') { steps { sh 'mvn test' } } } } } }



pipeline {
    agent any
    stages {
        stage('parallel tasks') {
            steps {
                sh 'mvn clean compile'
            }
        }
        stage('test') {
            steps {
                sh 'mvn test'
            }
        }
    }
}
