#2️⃣ Ansible — Install Nginx & Deploy HTML File


---
- name: Mini Project Web Server Deployment
  hosts: all
  become: yes
  tasks:
    - name: Install Nginx
      yum:
        name: nginx
        state: present

    - name: Start and enable Nginx
      service:
        name: nginx
        state: started
        enabled: yes

    - name: Deploy custom HTML index page
      copy:
        dest: /usr/share/nginx/html/index.html
        content: |
          <!DOCTYPE html>
          <html>
          <head><title>Pathnex Mini Project</title></head>
          <body><h1>Welcome to Pathnex DevOps Mini Project Deployment Live!</h1></body>
          </html>




#3️⃣ Kubernetes — Create Deployment + Service



apiVersion: apps/v1
kind: Deployment
metadata:
  name: project-nginx-deployment
spec:
  replicas: 3
  selector:
    matchLabels:
      app: web-app
  template:
    metadata:
      labels:
        app: web-app
    spec:
      containers:
        - name: nginx-server
          image: nginx:latest
          ports:
            - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: project-nginx-service
spec:
  type: LoadBalancer
  ports:
    - port: 80
      targetPort: 80
  selector:
    app: web-app




#4️⃣ Shell — Print CPU, Memory, Disk


#!/bin/bash
echo "=== SYSTEM MONITORING DASHBOARD ==="

# Get 1-minute CPU Load Average
CPU_LOAD=$(uptime | awk -F'load average:' '{print $2}' | awk '{print $1}' | sed 's/,//')
echo "CPU Load Average (1 min): $CPU_LOAD"

# Get Free/Used Memory metrics
echo "--- Memory Metrics ---"
free -h

# Get Disk Space Usage
echo "--- Disk Space Usage ---"
df -h /

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    triggers {
        cron('H 2 * * *') // Triggers everyday automatically at 2:00 AM
    }
    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
    }
}
