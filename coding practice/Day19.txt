- name: deploy app using Git
  hosts: all
  become: yes
  tasks:
    - name: Install git 
      yum:
        name: git
        state: present
    - name: clone repository
      git:
        repo: "https://github.com/pathnex/app.git"
        dest: "/opt/pathnex-app:"

#🔹 Kubernetes — Deployment Rolling Update (Detailed)
#apiVersion: apps/v1 kind: Deployment metadata: name: pathnex-rollout 
#spec: replicas: 3 strategy: type: RollingUpdate rollingUpdate: maxSurge: 2 
#maxUnavailable: 1 selector: matchLabels: app: rollout template: metadata:
# labels: app: rollout spec: containers: - name: app image: nginx


apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-rollout
spec:
  replicas: 3
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 2
      maxUnavailable:1
  selector:
    matchLabels:
    app: rollout
  template:
    metadata:
      labels:
        app: rollout
    spec:
      containers:
      - name: app
        iamge: nginx

    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"

            steps {

                sh 'echo "Deployment approved and executed!"'
            }
        }
    }
}
